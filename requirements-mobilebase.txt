# reachy_sdk_api, grpcio, protobuf, numpy, cv2 from ~/dev):
#   python3 -m venv --system-site-packages .venv-mobilebase
#   .venv-mobilebase/bin/pip install -r requirements-mobilebase.txt
#
# pip skips anything the system already provides. Do NOT add packages here that
# require protobuf>=4.
